  TimeShareChart, KLineChart }
  from '@libra-design/react'

function App() {
  const [stock, setStock] = useState(null)
  return (
    <>
      <StockTable data={data}
        onRowClick={(r) => setStock(r)}
        columnPicker navigable
        selectable />

        onClose={() => setStock(null)}>

    </>
